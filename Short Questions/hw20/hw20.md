## 3. Compare `let` vs `var`, explain with examples

```js
// var: function-scoped, can be redeclared
var x = 10;
if (true) {
    var x = 20;
    console.log(x); // 20
}
console.log(x); // 20

// let: block-scoped, can't be redeclared in the same scope
let y = 10;
if (true) {
    let y = 20;
    console.log(y); // 20
}
console.log(y); // 10
```

## 7. HTML page for lucky number (Placeholder)
*Implement an HTML page with an input box and button. On click or enter, generate a lucky number.*
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>Lucky Number Generator</title>
</head>
<body>
  <h1>Lucky Number Generator</h1>

  <input
    id="nameInput"
    type="text"
    placeholder="Enter your name (optional)"
  />
  <button id="generateBtn">Get Lucky Number</button>

  <div id="lucky-number"></div>

  <script>
    function generateLuckyNumber() {
      const name = document.getElementById("nameInput").value.trim();
      const luckyNum = Math.floor(Math.random() * 100) + 1;
      const display = name
        ? `${name}, your lucky number is: ${luckyNum}`
        : `Your lucky number is: ${luckyNum}`;
      document.getElementById("lucky-number").textContent = display;
    }

    // DOM Ready
    document.addEventListener("DOMContentLoaded", function () {
      document
        .getElementById("generateBtn")
        .addEventListener("click", generateLuckyNumber);

      document
        .getElementById("nameInput")
        .addEventListener("keypress", function (event) {
          if (event.key === "Enter") {
            generateLuckyNumber();
          }
        });
    });
  </script>
</body>
</html>
```

## 8. HTML page to fetch GitHub repos (Placeholder)
*Create an HTML page with a text box and button. Fetch GitHub repos via: `https://api.github.com/users/{username}/repos`.*
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>GitHub Repositories Fetcher</title>
  <style>
    body {
      font-family: Arial, sans-serif;
      padding: 50px;
      text-align: center;
      background-color: #f5f5f5;
    }

    input, button {
      padding: 10px;
      font-size: 16px;
      margin: 10px;
    }

    #repoList {
      margin-top: 30px;
      text-align: left;
      max-width: 600px;
      margin-left: auto;
      margin-right: auto;
    }

    li {
      margin-bottom: 8px;
    }
  </style>
</head>
<body>
  <h1>🔍 GitHub Repo Fetcher</h1>

  <input type="text" id="usernameInput" placeholder="Enter GitHub username" />
  <button id="fetchBtn">Fetch Repos</button>

  <ul id="repoList"></ul>

  <script>
    async function fetchRepos() {
      const username = document.getElementById("usernameInput").value.trim();
      const list = document.getElementById("repoList");
      list.innerHTML = "";

      if (!username) {
        list.innerHTML = "<li>Please enter a GitHub username.</li>";
        return;
      }

      const url = `https://api.github.com/users/${username}/repos`;

      try {
        const response = await fetch(url);
        if (!response.ok) throw new Error("User not found");

        const repos = await response.json();

        if (repos.length === 0) {
          list.innerHTML = "<li>No public repositories found.</li>";
        } else {
          repos.forEach(repo => {
            const li = document.createElement("li");
            li.innerHTML = `<a href="${repo.html_url}" target="_blank">${repo.name}</a>`;
            list.appendChild(li);
          });
        }
      } catch (error) {
        list.innerHTML = `<li>Error: ${error.message}</li>`;
      }
    }

    document.getElementById("fetchBtn").addEventListener("click", fetchRepos);
    document.getElementById("usernameInput").addEventListener("keypress", function (e) {
      if (e.key === "Enter") fetchRepos();
    });
  </script>
</body>
</html>
```

## 9. Explain JavaScript async non-blocking features
### Event Loop, Macro-task, Micro-task
JavaScript uses an event loop to handle asynchronous operations. It has two types of tasks: macro-tasks (like setTimeout, setInterval) and micro-tasks (like promises). The event loop processes the micro-tasks queue before the macro-tasks queue, allowing for non-blocking execution.
