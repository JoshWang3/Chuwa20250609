### 2. Resolve 5 leetcode problems using Javascript.

1. Two Sum

```javascript
/**
 * @param {number[]} nums
 * @param {number} target
 * @return {number[]}
 */
var twoSum = function (nums, target) {
        const map = new Map();
        for (let i = 0; i < nums.length; i++) {
            const complement = target - nums[i];

            if (map.has(complement)) {
                return [map.get(complement), i];
            }

            map.set(nums[i], i);
        }

    };
```

3. Longest Substring Without Repeating Characters

```javascript
/**
 * @param {string} s
 * @return {number}
 */
var lengthOfLongestSubstring = function (s) {
        const lastSeen = new Map(); // char -> last index
        let left = 0;               // start of window
        let maxLen = 0;

        for (let right = 0; right < s.length; right++) {
            const ch = s[right];

            // If we've seen this char inside the current window, move left past it
            if (lastSeen.has(ch) && lastSeen.get(ch) >= left) {
                left = lastSeen.get(ch) + 1;
            }
            lastSeen.set(ch, right);
            maxLen = Math.max(maxLen, right - left + 1);
        }
        return maxLen;
    };
```

5. Longest Palindromic Substring

```javascript
/**
 * @param {string} s
 * @return {string}
 */
var longestPalindrome = function (s) {
        if (!s || s.length < 2) return s;

        let start = 0, end = 0;

        const expand = (l, r) => {
            while (l >= 0 && r < s.length && s[l] === s[r]) {
                l--;
                r++;
            }
            // returns [leftIdx, rightIdx] of the palindrome AFTER expansion
            return [l + 1, r - 1];
        };

        for (let i = 0; i < s.length; i++) {
            // Odd-length center at i
            let [l1, r1] = expand(i, i);
            if (r1 - l1 > end - start) {
                start = l1;
                end = r1;
            }

            // Even-length center between i and i+1
            let [l2, r2] = expand(i, i + 1);
            if (r2 - l2 > end - start) {
                start = l2;
                end = r2;
            }
        }

        return s.slice(start, end + 1);
    };
```

9. Palindrome Number

```javascript
/**
 * @param {number} x
 * @return {boolean}
 */
var isPalindrome = function (x) {
        if (x < 0 || (x % 10 === 0 && x !== 0)) return false;

        let reversed = 0;
        let original = x;

        // Reverse half of the number
        while (x > reversed) {
            reversed = reversed * 10 + (x % 10);
            x = Math.floor(x / 10);
        }

        // Palindrome if left half == reversed right half
        // For odd lengths, remove middle digit with Math.floor(reversed / 10)
        return x === reversed || x === Math.floor(reversed / 10);
    };
```

14. Longest Common Prefix
```javascript
/**
 * @param {string[]} strs
 * @return {string}
 */
var longestCommonPrefix = function(strs) {
          if (!strs.length) return "";

          // Start with the first string as the prefix
          let prefix = strs[0];

          for (let i = 1; i < strs.length; i++) {
            // Shrink prefix until it matches the start of strs[i]
            while (strs[i].indexOf(prefix) !== 0) {
              prefix = prefix.slice(0, -1);
              if (!prefix) return "";
            }
          }
          return prefix;
        };
```

### 3. Compare let vs var , explain variable hosting with your own code examples.

- **`var`** → function-scoped, hoisted & initialized to `undefined`, redeclaration allowed.
- **`let`** → block-scoped, hoisted but **not** initialized (TDZ), no redeclaration in the same scope.
- Use `let` (or `const`) for safer, more predictable code.

#### Hoisting with `var`

```javascript
console.log(a); // ✅ undefined (hoisted & initialized)
var a = 5;
console.log(a); // ✅ 5
```

#### Hoisting with `let`

```javascript
// b is hoisted but uninitialized (Temporal Dead Zone)
console.log(b); // ❌ ReferenceError (Temporal Dead Zone)
let b = 10;
console.log(b); // ✅ 10
```

### 4. Explain closure with code example

- A closure is created when a function remembers variables from its outer scope, even after that outer function has
  finished executing.
- It’s like a backpack 🎒 that the function carries around — keeping the variables it needs alive.
- Example:

```javascript
function outerFunction() {
    let count = 0; // variable in outer scope

    function innerFunction() {
        count++;
        console.log(count);
    }

    return innerFunction;
}

const counter = outerFunction(); // outerFunction runs, returns innerFunction

counter(); // 1
counter(); // 2
counter(); // 3

```

**How this works:**

1. `outerFunction` runs and creates `count = 0`.
2. It **returns** `innerFunction` — but `innerFunction` still has access to `count` (thanks to the closure).
3. Even though `outerFunction` has finished, `count` stays alive in memory because `innerFunction` still uses it.
4. Every time you call `counter()`, it **remembers** the last value of `count`.

### 6. Explain Promise, Async, Await with code examples.

#### **1\. Promise**

A **Promise** represents a value that will be available **now**, **later**, or **never** (if it fails).

```javascript
const myPromise = new Promise((resolve, reject) => {
    setTimeout(() => resolve("Done!"), 1000);
});

myPromise.then(result => console.log(result)) // Done!
    .catch(err => console.error(err));
```

- **`then()`** → runs when resolved
- **`catch()`** → runs when rejected

---

#### **2\. Async**

Mark a function `async` to make it **always return a Promise**.

```javascript
async function greet() {
    return "Hello"; // wraps into Promise.resolve("Hello")
}

greet().then(msg => console.log(msg)); // Hello
```

---

#### **3\. Await**

Use `await` **inside async functions** to pause until a Promise settles.

```javascript
async function getData() {
    const data = await new Promise(res => setTimeout(() => res("Loaded"), 1000));
    console.log(data);
}

getData(); // waits 1 sec → "Loaded"
```

---

✅ **In short:**

- **Promise** = future value container
- **async** = makes a function return a Promise
- **await** = pauses inside async function until Promise resolves

### 9. Explain how Javascript implement asynchronous non-blocking feature. Particularly: Event Loop, Macrotask, and Microtask with code samples.

#### 1\. **JavaScript’s Nature**

- JavaScript itself is **single-threaded** (one call stack).
- Non-blocking happens because browsers/Node have background helpers (Web APIs) that do work and send results back
  later.

#### 2\. **Event Loop**

- Keeps checking: Is the main stack empty?
- If yes → takes the next task from a queue and runs it.
- Always runs microtasks before the next big task.

#### 3\. **Macrotask vs Microtask**

- Macrotask (big tasks): setTimeout, setInterval, DOM events, I/O.
- Microtask (small fast tasks): Promise.then, queueMicrotask.
- After each macrotask, all microtasks run before the next one.

```javascript
console.log("Start");

setTimeout(() => console.log("Macrotask"), 0);

Promise.resolve().then(() => console.log("Microtask"));

console.log("End");

//Start
//End
//Microtask
//Macrotask
```

- Sync code runs first.
- Then microtasks.
- Then macrotasks.
